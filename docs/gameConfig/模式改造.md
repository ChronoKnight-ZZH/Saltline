### 1.4 简报屏幕状态与逻辑（`createMenus` 内）

在 `let p = "none", d = null;` 之后：

```js
  // M5: briefing screen state
  let briefing = { lane: null, cell: null };
  let briefingEnabled =
    !(context.params && (context.params.time || context.params.weather ||
      context.params.shot || context.params.autostart));

  function pickCell(laneKey) {
    let cells = LANES[laneKey].cells;
    return cells[Math.floor(Math.random() * cells.length)];
  }

  function renderBriefing() {
    let text = o(".mb-brief-text");
    if (!text) return;
    if (!briefing.lane || !briefing.cell) {
      text.textContent = "";
      return;
    }
    text.textContent = briefText(briefing.cell[0], briefing.cell[1], briefing.lane);
  }

  function syncLaneButtons() {
    for (let b of r(".lane-btn")) {
      b.setAttribute("aria-pressed", String(b.dataset.lane === briefing.lane));
    }
  }

  function selectLane(laneKey) {
    if (!LANES[laneKey]) return;
    briefing.lane = laneKey;
    briefing.cell = pickCell(laneKey);
    syncLaneButtons();
    renderBriefing();
    o('[data-act="reroll"]').disabled = false;
    o('[data-act="deploy-lane"]').disabled = false;
  }

  function rerollLane() {
    if (!briefing.lane) return;
    briefing.cell = pickCell(briefing.lane);
    renderBriefing();
  }

  function showBriefing() {
    briefing.lane = null;
    briefing.cell = null;
    syncLaneButtons();
    renderBriefing();
    let rr = o('[data-act="reroll"]');
    let dp = o('[data-act="deploy-lane"]');
    if (rr) rr.disabled = true;
    if (dp) dp.disabled = true;
    q.show("briefing");
  }

  function deployLane() {
    if (!briefing.lane || !briefing.cell) return;
    if (context.environment && context.environment.set) {
      context.environment.set({
        timeOfDay: briefing.cell[0],
        weather: briefing.cell[1],
      });
    }
    context.events.emit("ui:click", {});
    e.deploy({
      lane: briefing.lane,
      timeOfDay: briefing.cell[0],
      weather: briefing.cell[1],
    });
  }
```

**要点**：
- `briefingEnabled` 用 `context.params` 判断，`runtime` 的 `params` 是 `Object.fromEntries(new URLSearchParams(location.search))`，是普通对象。
- `q` 是 `createMenus` 返回的对象，`q.show("briefing")` 切换屏幕；`q` 在 `createMenus` 末尾定义，`showBriefing` 在 `q` 之前定义，但**函数体在调用时执行**，`q` 那时已存在。**需把 `showBriefing` 等函数定义放在 `q` 之后**，或把 `showBriefing` 内改为 `s.dataset.screen = "briefing"; s.classList.add("open");` 直接操作 DOM，避免 `q` 依赖。

**推荐**：`showBriefing` 直接操作 DOM（与 `q.show` 等价）：

```js
  function showBriefing() {
    briefing.lane = null;
    briefing.cell = null;
    syncLaneButtons();
    renderBriefing();
    let rr = o('[data-act="reroll"]');
    let dp = o('[data-act="deploy-lane"]');
    if (rr) rr.disabled = true;
    if (dp) dp.disabled = true;
    p = "briefing";
    s.dataset.screen = "briefing";
    s.classList.add("open");
    S(); // 关闭设置面板（若开着）
    setTimeout(H, 30); // 聚焦主按钮
  }
```

### 3.2 暂停时更新 "Mission profile" 行

`flow.js` 的 `pause()`（或 `update` 内 `state === "paused"` 时）调：

```js
      if (context.environment && context.menus && context.menus.setProfile) {
        context.menus.setProfile(
          profileText(
            context.environment.timeOfDay,
            context.environment.weather,
            // lane 从 payload 或 environment 推断
            payload?.lane || inferLane(context.environment.timeOfDay, context.environment.weather),
          ),
        );
      }
```

**`inferLane`**：`flow.js` 内加：

```js
function inferLane(timeOfDay, weather) {
  if (weather === "heavyRain") return "assault";
  if (timeOfDay === "midnight" && (weather === "clear" || weather === "overcast")) return "stealth";
  if ((timeOfDay === "dawn" || timeOfDay === "noon") && (weather === "clear" || weather === "overcast")) return "recon";
  return "balanced";
}
```

**推荐**：`flow.start(payload)` 把 `payload.lane` 存到 `run.lane`，`pause()` 时用 `run.lane`；若 `run.lane` 不存在（URL 启动），用 `inferLane`。

```js
      start(payload) {
        if (payload && payload.timeOfDay && payload.weather) {
          context.environment?.set({ ... });
        }
        run.lane = payload?.lane || inferLane(
          context.environment?.timeOfDay,
          context.environment?.weather,
        );
        // ...
      },
      pause() {
        // ... 现有
        context.menus?.setProfile?.(profileText(
          context.environment?.timeOfDay,
          context.environment?.weather,
          run.lane,
        ));
      },
```

`flow.js` 需 `import { profileText } from "../ui/menus.js";`（或 `menus.js` 的 `q.setProfile` 接收 `lane`，`profileText` 在 `menus.js` 内生成）。**M5 决策**：`profileText` 在 `menus.js` 内导出，`flow.js` import。

### 8.5 简报 TTS（§9 追加）

DEPLOY 时用浏览器 `speechSynthesis` 朗读简报台词。

| 项 | 值 |
|---|---|
| 触发点 | `deployLane()` 内、`e.deploy()` 之前，**同步**调用 |
| 文本 | `briefText(cell[0], cell[1], lane)`，与屏幕显示完全一致 |
| 语音 | `lang = en-US`、`rate = 0.95`、`pitch = 0.9`、`volume = 0.9` |
| 语音挑选 | 本地英文（`localService`）→ 任意英文 → 不设 `voice`（浏览器默认） |
| 队列 | 每次 `speak()` 前先 `cancel()`，快速连点不叠音 |
| 失败兜底 | 不支持 / 抛错 / 空文本一律静默返回，**绝不阻断部署** |

**为什么必须放在 `deployLane()` 里**：浏览器要求 `speechSynthesis.speak()` 处于用户手势的回调链内，否则 Chrome 静默忽略（不报错、不发声）。`deployLane` 由 click 处理器直接调用，满足条件；一旦挪进 `setTimeout`、Promise 或 `flow.start` 之后的异步回调就会失效。

**与下方 §1.1 代码片段的差异**：片段把 `speakBriefing` 写在 `deployLane` 内部，并假设该函数里有 `environment.set`。实际 `environment.set` 已上移到 `flow.start`（§8.1），所以 `speakBriefing` 放在模块级（与 `briefText` 同级）以便复用和单测，调用点仍在 `deployLane` 内。

**voices 异步加载**：`getVoices()` 首次常返回 `[]`。处理方式：
1. `showBriefing()` 时调 `warmVoices()` 挂 `voiceschanged`（兼容 `addEventListener` 与老式 `onvoiceschanged` 属性）；
2. `collectVoices()` **每次调用都重新查** `getVoices()`，不依赖事件是否派发；
3. 仍拿不到就退回默认语音，不阻塞。

**测试覆盖（33 项，临时脚本已删）**：无 API 时静默且部署正常、文本/语速/音调/音量精确、优先本地英文、只有中文时退回默认、非本地英文也接受、连点两次 `cancel` 生效且发两条 payload、REROLL 与选卡**不发声**、`speak` 抛错不影响开局、`voiceschanged` 后能补选到英语音、空/`null` 文本不播报。

**未验证**：实际发声效果（音色、断句、延迟）无法在无头环境验证，需真人试听。另外 Safari/iOS 对 `SpeechSynthesisUtterance` 的默认音量与 `cancel()` 行为有差异，可能需要在实机上微调 `volume`。
