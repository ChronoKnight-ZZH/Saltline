### 3.1 导出与结构

### 3.2 内部状态

```js
state = {
  enabled: false,          // 由 environment._applyStatic 控制
  active: false,           // enabled && 有光柱
  pos: new Vector3(),
  vel: new Vector3(),
  yaw: 0,
  orbitA: 0,               // 盘旋角度
  orbitR: HELICOPTER_TUNING.radius,  // 18

  light: null,             // SpotLight（low 画质也保留）

  decal: null,             // 地面光斑贴花（low 必须有）
```

### 3.3 `setEnabled(on)`

```js
function setEnabled(on) {
  if (state.enabled === on) return;
  state.enabled = on;
  if (!on) {
    // 立即停：光柱隐藏、音效停、暴露清、光斑隐藏
    state.active = false;
    if (state.cone) state.cone.visible = false;
    if (state.decal) state.decal.visible = false;
    if (state.light) state.light.intensity = 0;
    if (state.loop) { try { state.loop.stop(); } catch {} state.loop = null; }
    state.lastSpot.clear();
    context.events?.emit("helicopter:spot", { targets: [] });
    return;
  }
  state.active = true;
  // 初始位置：玩家周围一圈
  state.orbitA = rng.range(0, Math.PI * 2);
  state.pos.set(
    player.pos.x + Math.cos(state.orbitA) * state.orbitR,
    state.height,
    player.pos.z + Math.sin(state.orbitA) * state.orbitR,
  );
  // 光柱方向：向下偏 yaw
  state.beamDir.set(0.18, -1, 0).normalize().applyAxisAngle(Y_AXIS, rng.range(0, Math.PI * 2));
  // 音效
  if (context.audio?.loop) {
    try { state.loop = context.audio.loop("helicopter", { pos: state.pos, gain: 0.7 }); } catch { state.loop = null; }
  }
}
```

**要点**：

- `setEnabled(false)` 立即清光柱/音效/暴露，**不等 damp**——改造文档 4.4“夜雨切换：立即停光柱 + 停音效 + 清暴露”。
- `setEnabled(true)` 从玩家周围随机角度开始，`rng` 派生自 `context.rng.fork()`，可复现。
- `helicopter:spot` 事件 M3 派发，HUD 订阅。

### 3.4 `update(dt)`

```js
function update(dt) {
  if (!state.active) return;
  // 1. 盘旋：玩家为中心的圆/椭圆
  state.orbitA += HELICOPTER_TUNING.speed * dt;
  const targetX = player.pos.x + Math.cos(state.orbitA) * state.orbitR;
  const targetZ = player.pos.z + Math.sin(state.orbitA) * state.orbitR;
  const targetY = state.height + Math.sin(state.orbitA * 2.3) * 1.5;
  // 平滑跟随（damp，避免瞬移）
  state.pos.x = damp(state.pos.x, targetX, 2, dt);
  state.pos.y = damp(state.pos.y, targetY, 2, dt);
  state.pos.z = damp(state.pos.z, targetZ, 2, dt);
  // yaw 朝向玩家
  state.yaw = Math.atan2(player.pos.x - state.pos.x, player.pos.z - state.pos.z);
  // 2. 光柱几何
  state.beamOrigin.copy(state.pos);
  state.beamOrigin.y -= 0.5;  // 机身下方
  if (state.light) {
    state.light.position.copy(state.beamOrigin);
    state.light.target.position.copy(state.beamOrigin).addScaledVector(state.beamDir, HELICOPTER_TUNING.beamRange);
    state.light.target.updateMatrixWorld();
  }
  if (state.cone) {
    // 锥体网格：origin + dir + halfAngle + range
    orientCone(state.cone, state.beamOrigin, state.beamDir, HELICOPTER_TUNING.beamRange, HELICOPTER_TUNING.beamHalfAngle);
  }
  if (state.decal) {
    // 地面光斑：光柱与地面的交点
    const hit = context.physics.raycast(state.beamOrigin, state.beamDir, HELICOPTER_TUNING.beamRange);
    if (hit) {
      state.decal.position.copy(hit.point);
      state.decal.visible = true;
    } else {
      state.decal.visible = false;
    }
  }
  // 3. 音效位置（10Hz / low 5Hz）
  state.audioT -= dt;
  const audioInterval = context.quality?.name === "low" ? 0.2 : 0.1;
  if (state.audioT <= 0) {
    state.audioT = audioInterval;
    if (state.loop) {
      state.loop.setPos(state.pos);
      state.loop.setGain(0.7 * (1 - (context.environment?.perception?.audioMask ?? 0) * 0.3));
    }
  }
  // 4. 暴露扫描
  state.spotT -= dt;
  if (state.spotT <= 0) {
    state.spotT = HELICOPTER_TUNING.exposeInterval;  // 0.25s（low 0.5s）
    scanAndExpose();
  }
}
```

**要点**：

- `damp` 来自 `core/math.js`，平滑跟随玩家。
- 光柱方向 `beamDir` M3 固定（随机初始方向 + 缓慢旋转可选）；M3 简化：固定向下偏，不旋转，降低调试复杂度。
- `audioMask` 雨声降低直升机音量（“雨声掩蔽”），与 `footstepMaskAt` 语义一致。
- `exposeInterval` 从 `HELICOPTER_TUNING.exposeInterval = 0.25`，low 画质×2 → 0.5。

### 3.5 `isInBeam(targetPos)`

```js
function isInBeam(targetPos) {
  if (!state.active) return false;
  // 1. 距离粗筛
  const toTarget = _v1.copy(targetPos).sub(state.beamOrigin);
  const dist = toTarget.length();
  if (dist > HELICOPTER_TUNING.beamRange + 0.3) return false;  // 0.3m 容差
  // 2. 锥体判定
  toTarget.multiplyScalar(1 / dist);  // 归一化
  const dot = toTarget.dot(state.beamDir);
  if (dot < Math.cos(HELICOPTER_TUNING.beamHalfAngle)) return false;
  // 3. LOS
  return context.physics.lineOfSight(state.beamOrigin, targetPos);
}
```

**要点**：

- 0.3m 容差：改造文档 4.1 第 2 步“锥体判定 0.3m 容差”。
- 零距离保护：`dist < 1e-6` 时直接返回 `false`（目标在光源原点，无意义）。
- `lineOfSight` 来自 `physics`，与 M4 闪电共用。

### 3.6 `getSpotTargets()`

```js
function getSpotTargets() {
  if (!state.active) return [];
  const targets = [];
  // 1. 敌人
  for (const enemy of context.enemies?.list || []) {
    if (!enemy.alive || !enemy.active) continue;
    if (isInBeam(enemy.pos)) targets.push({ type: "enemy", ref: enemy });
  }
  // 2. 玩家
  if (context.player?.alive !== false) {
    const eye = context.player.eye || context.player.pos;
    if (isInBeam(eye) || isInBeam(context.player.pos)) {
      targets.push({ type: "player", ref: context.player });
    }
  }
  return targets;
}
```

**要点**：

- 玩家判定 `eye` 和 `pos` 两个点（站/蹲），覆盖更全。
- 返回 `{type, ref}`，M3 的 `scanAndExpose` 消费。

### 3.7 `scanAndExpose()`（暴露扫描，改造文档 4.1）

```js
function scanAndExpose() {
  // 0. 粗筛：光柱外接球 vs WORLD_BOUNDS
  if (!beamIntersectsWorld()) return;
  // 1. 收集光柱内目标
  const targets = getSpotTargets();
  if (!targets.length) {
    // 无目标也要派发（HUD 清空）
    context.events?.emit("helicopter:spot", { targets: [] });
    return;
  }
  // 2. 对称暴露
  const enemyIds = [];
  for (const t of targets) {
    if (t.type === "enemy") {
      const enemy = t.ref;
      // 敌人被照到：写 lastKnown + investigate
      context.enemies.forceExpose(context.player.pos, "helicopter", enemy);
      enemyIds.push(enemy.id);
    } else if (t.type === "player") {
      // 玩家被照到：所有 LOS 可达玩家的敌人 forceExpose
      for (const enemy of context.enemies?.list || []) {
        if (!enemy.alive || !enemy.active) continue;
        if (enemy.mode === "engage") continue;  // 已交战，不降级
        if (context.physics.lineOfSight(enemy.sensor || enemy.pos, context.player.eye || context.player.pos)) {
          context.enemies.forceExpose(context.player.pos, "helicopter", enemy);
          enemyIds.push(enemy.id);
        }
      }
    }
  }
  // 3. 派发事件（HUD 订阅）
  context.events?.emit("helicopter:spot", {
    targets: enemyIds.map((id) => ({ id })),  // 只传 id，HUD 自己查 enemy
    playerExposed: targets.some((t) => t.type === "player"),
  });
}
```

**要点**：

- 粗筛 `beamIntersectsWorld()`：光柱外接球（`beamOrigin + beamDir * range/2`，半径 `range/2`）与 `WORLD_BOUNDS` AABB 相交。简单实现：`beamOrigin.y > WORLD_BOUNDS.max.y + range` 或 `beamOrigin.y < WORLD_BOUNDS.min.y - range` 时 return；或更精确的球-AABB 相交。
- 敌人被照到：`forceExpose(player.pos, "helicopter", enemy)` —— 敌人知道玩家位置，即使玩家不在敌人视觉半径内（光柱是信息源）。
- 玩家被照到：所有 LOS 可达玩家的敌人 `forceExpose` —— 对称。
- `enemy.mode === "engage"` 跳过：改造文档 4.5“AI：只写 `lastKnown` + 触发 `investigate`，不强制 `engage`”；已交战的不降级。
- 事件 payload：`{ targets: [{id}], playerExposed }`，HUD 订阅后按 `id` 查 `enemies.list` 拿 `enemy`。

### 3.8 光柱渲染（quality-dependent）

| 画质 | `envVolumetricBeam` | 实现 |
|---|---|---|
| `ultra/high` | `true` | 锥形网格 + 噪声 + additive blending |
| `medium` | `false` | 纯半透明锥形网格（`MeshBasicMaterial` + `transparent` + `opacity`） |
| `low/test` | `false` | 仅 `SpotLight` + 地面光斑贴花（`CircleGeometry` + `MeshBasicMaterial`） |

**M3 实现顺序**：先 low（SpotLight + 贴花，最简单），再 medium（半透明锥体），最后 ultra/high（噪声锥体）。M3 验收时至少 low + medium 可跑。

`SpotLight` 创建：
```js
if (!state.light) {
  state.light = new SpotLight(0xfff2c8, 8, HELICOPTER_TUNING.beamRange, HELICOPTER_TUNING.beamHalfAngle, 0.35, 1.2);
  state.light.castShadow = false;  // 阴影太贵，M3 不开
  context.scene.add(state.light);
  context.scene.add(state.light.target);
}
```

**要点**：

- `castShadow = false`：光柱不需要投影，M3 不开。
- `angle = beamHalfAngle`，`penumbra = 0.35`，`decay = 1.2`。
- 颜色 `0xfff2c8`（暖白），强度 8，M6 调。

### 4.2 `src/enemies/enemies.js`

**4.2.1 新增 `forceExpose(pos, mode, targetEnemy?)`**

```js
      forceExpose(pos, mode, targetEnemy) {
        // mode: "helicopter" | "lightning"
        // targetEnemy: 若提供，只对该敌人生效；否则对所有 LOS 可达的敌人
        const now = h.t;
        const cooldown = 1.5;
        const targets = targetEnemy ? [targetEnemy] : t;
        for (const enemy of targets) {
          if (!enemy.alive || enemy.passive) continue;
          if (enemy.mode === "engage") {
            // 已交战：只更新 lastKnown，不降级
            enemy.lastKnown.copy(pos);
            continue;
          }
          const last = h.exposeCooldown.get(enemy.id) || -9;
          if (now - last < cooldown) continue;
          h.exposeCooldown.set(enemy.id, now);
          enemy.lastKnown.copy(pos);
          enemy.mode = "investigate";
          enemy.modeT = Math.max(2, enemy.modeT || 0);
          enemy.senseT = 0;
          context.events.emit("enemy:alert", { enemy, noise: pos });
        }
      },
```

**要点**：

- `h.exposeCooldown = new Map()` 在 `createEnemies` 内初始化。
- `modeT = max(2, ...)`：改造文档 4.4“进入 `investigate` 后最短 2s”。
- `senseT = 0`：强制下一帧重算 LOS。
- `enemy:alert` 事件复用现有（`{enemy, noise}`），HUD/音频可订阅。
- 1.5s 冷却在 `enemies` 侧，HUD 侧另算（见 4.4）。

**4.2.3 `alert` 与 `forceExpose` 的关系**

```js

  await initializeSystem("environment", "environment", () => createEnvironment(context));
  await initializeSystem("helicopter", "helicopter", () => createHelicopter(context));
  // 然后 URL 参数处理
  if (params.time || params.weather) { ... }
  else if (params.shot || params.bench) { ... }

  await setBootProgress("Waiting for first light", 0.92);
```

**关键**：URL 参数处理必须在 `helicopter` 初始化之后，否则 `set/randomize` 时 `context.helicopter` 不存在，`_applyStatic` 跳过启停。

## 5. 暴露扫描实现细节（改造文档 4.1 对照）

| 改造文档步骤 | M3 实现 |
|---|---|
| 1. 光柱外接球 vs WORLD_BOUNDS 相交粗筛，无交集 return | `beamIntersectsWorld()`：`beamOrigin` 到 `WORLD_BOUNDS` 的球-AABB 相交；M3 简化为“`beamOrigin.y` 在 `[min.y - range, max.y + range]` 之外 return” |
| 2. 遍历敌人：AABB vs 光柱外接球 → 锥体判定（dot + 距离）→ LOS | `getSpotTargets()` + `isInBeam()`：距离粗筛（`dist > range + 0.3` return）→ 锥体（`dot < cos(halfAngle)` return）→ LOS |
| 3. 通过者写入 exposureSet：HUD 标记（上限 3，按距离排序）+ 敌人 `lastKnown` + `investigate` | `scanAndExpose()`：敌人被照到 → `forceExpose(player.pos, "helicopter", enemy)`；派发 `helicopter:spot`，HUD 订阅 |
| 4. 同时判定玩家是否在光柱内 + 敌人 LOS 可达玩家 → 反之暴露玩家 | `scanAndExpose()`：玩家被照到 → 遍历敌人，LOS 可达 → `forceExpose` |

**状态机稳定性（4.4）**：

| 规则 | M3 实现 |
|---|---|
| 光柱边缘：锥体判定加 0.3m 容差 | `isInBeam` 的 `dist > range + 0.3` 和 `dot < cos(halfAngle)` 不加容差（锥体判定不加重），距离粗筛加 0.3m |
