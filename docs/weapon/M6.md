# M6 修改方案：武器系统与地图弹药补给

> **当前状态（2026-11-15）：M6.0 地图弹药补给已落地并真机验证通过**，改动 10 个文件（4 新 / 6 改），`npm run check` 80 模块通过、`npm run build` 通过。落地细节与实测数据见 §3.4.6，落点见 §3.4.1。M6.1–M6.7（武器数据化、切枪、选枪页、HUD 槽条、多弹丸）尚未开始。
>
> 依据：`docs/武器与弹药补给系统设计.docx`（v1.0，2026-11-15，`feature/weapon-ammo`）+ 当前 `src/` 实测。
>
> 结论先行：设计稿的数值全部能落在现有接口上，但有 **5 处代码硬约束必须先改**：
> 1. 武器数值是模块级常量（`weapon.js:29-37`），不是可替换的配置；
> 2. `combat.playerShot` 是单射线、`stats.shots` 每发 +1（`combat.js:18-29`），霰弹的多弹丸无法直接表达；
> 3. 交互提示只有"按住 E"一种形态（`hud.js:347`），补给要的是"轻按 E"；
> 4. 关卡静态网格会整体合批（`level.js:126`），补给箱必须走 dynamic，否则变灰/光晕改不动；
> 5. `weapon.addAmmo(n)` 是裸加法、无上限（`weapon.js:445`），钳制必须由补给侧做，否则 PAC-9 会被补到 `reserve > reserveMax`。

### 3.2 `src/weapons/weapon.js` 参数化

改造原则：**只把常量换成 `def` 字段，不动动画与弹簧结构**，保证 `?show=weapon` 的观感不回退。

```js
async function createWeapon(context, def) {
  def = def || WEAPONS[DEFAULT_LOADOUT];
  ...
  let ve = {
    name: def.name,
    def,
    slot: def.slot,
    ammoType: def.ammoType,
    ammo: def.magSize,
    magSize: def.magSize,
    reserve: Math.min(def.reserveStart, def.reserveMax),
    ...
    addAmmo(n, cap = def.reserveMax) {
      ve.reserve = Math.min(cap, ve.reserve + Math.max(0, +n || 0));
      return ve.reserve;
    },
    ammoRoom: () => def.reserveMax - ve.reserve,
  };
```

扩散与天气的合成（保持现有"环境倍率只作用于弹道、不写回 `ve.spread`"约定，`weapon.js:550-557`）：

```js
function shotSpread(def, adsE, wet) {
  let base = lerp(def.spreadHip, def.spreadAds, adsE);
  let env = (context.environment && context.environment.perception.spreadScale) || 1;
  // 只对雨天与暴雨生效：晴/阴 spreadScale 本来就是 1.0，无需再判
  let rain = env > 1 ? (def.wetMul || 1) : 1;
  return base * env * rain;
}
```

// Weapon switching, per-weapon ammo and the context.weapon contract.
async function createWeaponSystem(context) {
  let byId = new Map(), order = WEAPON_ORDER.slice(), active = null, switching = 0;
  let pending = null;               // { id, t, phase: "down" | "up" }

  async function build(def) {

    w.setVisible(false);

    select(DEFAULT_LOADOUT, { immediate: true });
  }

  function cycle(dir) {
    let i = order.indexOf(active.id);
    select(order[(i + dir + order.length) % order.length]);
  }

  return {
    get list() { return order.map((id) => byId.get(id)); },
    get activeId() { return active && active.id; },
    select(id, { immediate = false } = {}) {
      let next = byId.get(id);
      if (!next || next === active) return false;

      return order.map((id) => {
        let w = byId.get(id);
        return { id, slot: WEAPONS[id].slot, name: WEAPONS[id].name,
                 short: WEAPONS[id].short, ammoType: WEAPONS[id].ammoType,
                 ammo: w ? w.ammo : 0, reserve: w ? w.reserve : 0,
                 active: !!active && active.id === id };
      });
    },
    ammoFor(type) { /* 聚合某口径总备弹，供 HUD/补给点判定 */ },
    fill(type, amount) { /* 对同口径每把枪 addAmmo(amount) */ },
    update(dt) { /* pending 状态机 + slot 输入 + context.weapon 指向 + player.tuning.fovAds */ },

落点算法（`level.js` 末尾，在 `_.forEach(E)` / 地面点吸附之后）：

```js
// 1.6m 环 16 方向的可行走数 = "这块地方够不够站人"
let fit = (x, z) => {
  let n = 0;
  for (let i = 0; i < 16; i++) {
    let a = i * 0.3927;
    M.isWalkable(x + Math.cos(a) * 1.6, z + Math.sin(a) * 1.6) && n++;
  }
  return n;
};
let R = SUPPLY_POINTS.map((s) => {
  let p = mt(s.x, 0, s.z), best = -1e9, bx = s.x, bz = s.z;
  // 6m 内螺旋：可行走才算候选，fit 高者优先，同分取离设计点最近的
  for (let r = 0; r <= 6; r += 0.5)
    for (let k = 0; k < (r ? 16 : 1); k++) {
      let a = k * 0.3927, x = s.x + Math.cos(a) * r, z = s.z + Math.sin(a) * r;
      if (!M.isWalkable(x, z)) continue;
      let score = fit(x, z) * 10 - r;
      score > best && ((best = score), (bx = x), (bz = z));
    }
  best > -1e9 && ((p.x = bx), (p.z = bz));
  console.warn("[level] supply " + s.id + " at (" + p.x.toFixed(2) + ", " + p.z.toFixed(2) +
    ") moved " + Math.hypot(p.x - s.x, p.z - s.z).toFixed(2) + "m, room " + fit(p.x, p.z) + "/16");
  let o = t.place("supply_crate", p.x, p.z, s.yaw, { seed: s.seed, kind: s.kind, id: s.id, label: "AMMO" }, 0, { dynamic: !0 });
  return { id: s.id, kind: s.kind, label: s.label, pos: mt(p.x, 0, p.z), object: o, used: !1 };
});
```

> 为什么不是"最近可走格吸附"：yard 里箱子是成排贴着的，最近可走格往往落在**箱缝里**（实测用旧的 `E()` 吸附后，箱子 1.6m 环上 8 个方向全部不可走，玩家走不到 2m 交互圈内）。按"能站人的空间"打分才有可玩性。
> 顺序仍然重要：**先算落点、后放置**。箱子自己的碰撞盒一旦进 `nav`，会把自己所在的格子标成不可走。

#### 3.4.2 新道具 `supply_crate`（M6.0 已落地）

新增 `src/world/props/supply-builders.js`，`register.js` 里在 `registerDetailedProps()` **之后**注册一行（因为它内部要调 `crate_wood`）：

```js
import { BoxGeometry, Mesh, MeshBasicMaterial, RingGeometry } from "../../vendor/three-r180.js";
import { PROP_BUILDERS } from "./registry.js";

var KIND_TINT = { mixed: 0xd8c07a, light: 0xd8a15c, heavy: 0x7fb3d5, shell: 0xd06a4a, dmr: 0x9fc6da };
var DARK = 0x4a4a4a;

function buildSupplyCrate(context, opts = {}) {
  let kind = opts.kind || "mixed",
    tint = KIND_TINT[kind] || KIND_TINT.mixed,
    group = PROP_BUILDERS["crate_wood"](context, {
      seed: opts.seed || 0,
      label: opts.label || "AMMO",
      code: "SLT-" + String(opts.id || "S").toUpperCase(),
    }),
    haloMaterial = new MeshBasicMaterial({ color: tint, transparent: true, opacity: 0.35, depthWrite: false }),
    halo = new Mesh(new RingGeometry(0.72, 1.12, 40), haloMaterial),
    lampMaterial = new MeshBasicMaterial({ color: tint }),
    lamp = new Mesh(new BoxGeometry(0.16, 0.05, 0.16), lampMaterial),
    bar = new Mesh(new BoxGeometry(0.62, 0.05, 0.09), lampMaterial),
    state = opts.used ? "used" : "live", t = 0;

  (halo.rotation.x = -Math.PI / 2), (halo.position.y = 0.018),
    (lamp.position.y = 1.02), (bar.position.y = 1.02),
    (halo.raycast = () => {}), (lamp.raycast = () => {}), (bar.raycast = () => {}),
    group.add(halo, lamp, bar);

  function setState(next) {
    state = next === "used" ? "used" : "live";
    (halo.visible = state !== "used"),
      lampMaterial.color.setHex(state === "used" ? DARK : tint);
  }
  function update(dt) {
    if (state === "used") return;
    t += dt;
    let p = context.environment && context.environment.perception,
      vis = (p && +p.visibility) || 45,
      glow = clamp01(1 - vis / 45),
      base = 0.35 + 0.4 * glow;
    haloMaterial.opacity = base * (0.55 + 0.45 * Math.sin(t * 2.4));
  }

  return ((group.userData.supply = { id: opts.id || "", kind }),
    (group.userData.setState = setState),
    (group.userData.update = update),
    setState(state), group);
}
```

要点：

- `PROP_BUILDERS["crate_wood"]` 是已注册的构建函数（`register.js` 里 `registerDetailedProps` 装的），这里**调用而不是 import 私有 `GA`**，避免把 `detailed-builders` 的私有符号暴露出去。
- `userData.update` 会被 `place()` 自动推进 `t.updaters`（`buildings.js:255`），脉冲动画不需要另开循环。
- 子网格一律 `raycast = () => {}`：光晕和灯不该吃子弹射线。
- 碰撞盒仍来自 `crate_wood` 的 `finish()`（`userData.colliders`，1.1×1×0.9），**叠加子网格不会改变碰撞盒**。
- 实测结论：`AdditiveBlending` 在正午强光下**看不见**（加色在灰白混凝土上过曝），改成普通混合后正午、午夜都清晰；光晕半径 0.72–1.12 才能露在箱体（±0.55 / ±0.45）之外。
- `used` 态：光晕 `visible=false`、指示灯变深灰；箱体材质是共享的（`mats` 缓存），**不能**整箱换色，否则 5 个箱子一起变灰。

#### 3.4.3 新增 `src/game/supply.js`（M6.0 已落地）

```js
import { DEFAULT_LOADOUT, WEAPONS } from "../weapons/definitions.js";

var SUPPLY_KINDS = {
  mixed: { types: ["pistol", "smg", "dmr", "shell"], mul: 0.5 },  // S1：四类各 50%
  light: { types: ["pistol", "smg"], mul: 1 },
  heavy: { types: ["dmr", "shell"], mul: 1 },
  shell: { types: ["shell"], mul: 1 },
  dmr: { types: ["dmr"], mul: 1 },
};
var RADIUS = 2, FACING = 0.45, VERTICAL = 2.5, OTHER_MUL = 0.3;

function createSupply(context) {
  let spots = (context.level.supplySpots || []).map((s) => ({ ...s }));

  // M6.0 只有 PAC-9；武器数 < 2 时按口径放行会让 S3/S4/S5 变成空箱子，
  // 所以单武器阶段一律按 mixed 处理，M6.1 起恢复按口径分计。
  function kindOf(spot) {
    return weapons().length < 2 ? SUPPLY_KINDS.mixed : SUPPLY_KINDS[spot.kind] || SUPPLY_KINDS.mixed;
  }
  // 当前武器实例还没有 def 字段（M6.1 才有），此时按 pac9 的定义取 reserveMax
  function defOf(w) { return (w && w.def) || WEAPONS[(w && w.id) || ""] || WEAPONS[DEFAULT_LOADOUT]; }

  function nearest(pos, forward) { /* 同 flow.$()：2m + 朝向点积 > 0.45，返回最近 spot */ }

  function usable(spot) {
    if (!spot || spot.used) return "used";
    if (context.solarflare && +context.solarflare.effect > 0) return "emp";
    return null;
  }

  function use(spot) {
    let why = usable(spot);
    if (why) return why;
    let kind = kindOf(spot), act = active(), matched = 0, filled = 0;
    for (let w of weapons()) {
      let def = defOf(w);
      if (!def || kind.types.indexOf(def.ammoType) < 0) continue;
      matched++;
      // 当前武器按 kind.mul 补（专用点 = 补满），其它同口径 +30%
      let add = Math.ceil(def.reserveMax * (w === act ? kind.mul : OTHER_MUL)),
        before = +w.reserve || 0,
        target = Math.min(def.reserveMax, before + add);   // addAmmo 本身无上限，钳制放在补给侧
      target > before && (w.addAmmo(target - before), (filled += target - before));
    }
    if (!matched) return "channel";     // 没有同口径武器：不消耗
    if (!filled) return "full";         // 已经满弹：不消耗补给点，只提示一次
    spot.used = true;
    spot.object && spot.object.userData.setState("used");
    context.events.emit("supply:used", { id: spot.id, kind: spot.kind, amount: filled });
    return null;
  }

  return { spots, kinds: SUPPLY_KINDS, nearest, usable, use,
    reset() { for (let s of spots) (s.used = false), s.object && s.object.userData.setState("live"); },
    remaining() { return spots.filter((s) => !s.used).length; } };
}
```

补给量口径（"补满"必须有上限，否则补给点等于无限弹药）：

| kind | 当前武器 | 其它武器（同 `types` 内） |
|---|---|---|
| `mixed`（S1） | `+ceil(reserveMax * 0.5)` | `+ceil(reserveMax * 0.3)` |
| 专用点（S2–S5） | `+ceil(reserveMax * 1.0)`（补满） | `+ceil(reserveMax * 0.3)` |

- 补的是**备弹**，不直接塞弹匣；弹匣仍由玩家按 R 换（保持换弹动画与音频节奏）。
- `full` / `channel` 返回值都不消耗补给点，flow 只提示一次 —— 避免玩家跑一趟发现箱子空了。
- `weapon.addAmmo(n)` 是裸加法（`weapon.js:445`，无上限），所以**钳制在补给侧做**：先算 `min(reserveMax, before + add)` 再 `addAmmo(delta)`。这样 M6.0 完全不动 `weapon.js`，PAC-9 行为零回退。

### 3.9 `src/combat/combat.js` — 多弹丸与距离衰减

```js
playerShot(origin, dir, {
  damage: dmg = 28, range = 300, impulse = 1,
  pellets = 1, pelletCone = 0, falloff = null, style = "pistol",
} = {}) {
  e.shots++;                                   // 一次扣扳机只算 1 发
  let best = null, anyHit = false;
  for (let i = 0; i < pellets; i++) {
    let d = i === 0 ? dir : perturb(dir, pelletCone);   // 第一颗走准心
    let out = castOne(origin, d);                        // 原 24-83 行逻辑抽成内部函数
    if (out && out.enemy) {
      anyHit = true;
      let mul = falloff ? falloffMul(falloff, out.dist) : 1;
      out.enemy.damage(dmg * (out.part === "weak" ? 2.2 : 1) * mul, ...);
      if (!best || out.dist < best.dist) best = out;
    }
    context.events.emit("impact", out);                  // 每弹丸都出弹着特效
  }
  if (anyHit) e.hits++;                                 // 命中率按"扣扳机次数"算
  best && context.events.emit("player:hitEnemy", ...);   // 只有最近一发打命中标记
  return best;
}
```

要点：

- **命中率分母**：`stats.shots` 每扣扳机 +1，不是每弹丸 +1，否则结算页命中率会崩（霰弹最多 800%）。
- **命中标记**：8 颗弹丸只出 1 次 `player:hitEnemy`，否则 HUD 命中音效叠 8 层。
- 弹着特效应逐颗出（`impact`），霰弹的墙面弹着才好看，但视觉上不触发 8 次 hitmarker。
- `perturb(dir, cone)` 用与武器相同的随机源，保证 `spread` 与弹丸扩散互不重复。
